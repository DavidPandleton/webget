```python
"""MCP server regression tests. Requires the mcp client SDK (dev extra)."""
import asyncio
import json
import os
import sys
from pathlib import Path

import pytest

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

ROOT = Path(__file__).resolve().parent.parent
MCP = ROOT / "webget_mcp.py"


@pytest.mark.parametrize(
    "tool_args",
    [
        {"url": "https://example.com", "strategy": "bogus"},
        {"url": "https://example.com", "strategy": "firecrawl"},
    ],
)
def test_invalid_strategy_does_not_kill_server(tool_args):
    async def run():
        params = StdioServerParameters(command=sys.executable, args=[str(MCP)])
        async with stdio_client(params) as (read, write), ClientSession(read, write) as session:
            await session.initialize()
            res = await session.call_tool("fetch", tool_args)
            return res

    res = asyncio.run(run())
    assert res.isError is False or "error" in res.content[0].text
    # server must still answer after the bad call
    async def alive():
        params = StdioServerParameters(command=sys.executable, args=[str(MCP)])
        async with stdio_client(params) as (read, write), ClientSession(read, write) as session:
            await session.initialize()
            res = await session.call_tool("fetch", {"url": "https://example.com", "strategy": "http", "no_cache": True})
            return res

    res2 = asyncio.run(alive())
    assert "success" in res2.content[0].text
```

NOTE: the two `asyncio.run` blocks spawn two separate server processes; that is fine and keeps the test deterministic. The first `call_tool` with `strategy: "bogus"` must NOT hang: it must return quickly with an error payload. Run with a hard timeout if needed: `pytest --timeout=60` (add `pytest-timeout` to dev deps or rely on the server-side guard).

    if strategy not in _VALID_STRATEGIES:

Keep two processes.

4. **Merge order**: PR #8 first or PR #7 first — no diff overlap (verified: both merge cleanly into main). Recommend merging #8 first (smaller, pure fix), then #7.
5. **No scope creep**: do NOT add features during this audit-fix pass (no search error-message polish, no timeout improvements). Those are follow-ups.
6. **Commit signature**: every commit ends with `- Rouge` (hyphen, per repo rule). Author email `tarigansdavid@gmail.com`.
